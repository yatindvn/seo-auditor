# Phase 3: Keywords Explorer + Content Gap Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Explore a seed keyword (Autocomplete + People Also Ask ideas, volume, difficulty, traffic estimate, optional SERP clusters) and compute a domain-level content gap ranked by opportunity, with CSV export. Provider responses are cached so a repeat costs nothing.

**Architecture:** A `KeywordDataProvider` interface with a DataForSEO adapter (search volume, bulk difficulty, ranked keywords) and a dry-run adapter. The DataForSEO adapter checks a DB cache first, then reserves from its own `dataforseo_labs` budget, then calls HTTP, so a cache hit costs nothing. The SERP adapter shares one `post_task` helper and now returns People Also Ask questions. Pure `explorer` and `gap` modules sit on top, and a FastAPI router exposes JSON and CSV.

**Tech Stack:** SQLAlchemy + Alembic, requests, FastAPI, csv, pytest.

**Spec:** `docs/superpowers/specs/2026-10-25-keywords-explorer-design.md`

## Global Constraints

- Branch `feat/keywords-explorer`. Baseline: backend 434 passed + 1 skipped.
- **No test touches the network.** Google Autocomplete is always faked in tests: `keyword_suggestions._autocomplete_suggestions` is looked up at call time so it can be monkeypatched.
- **Order of work in a DataForSEO keyword call:** cache → `spend.reserve("dataforseo_labs", DATAFORSEO_LABS_DAILY_CAP)` → HTTP. A hit makes no request and reserves nothing.
- **Dry-run results are never cached and are labelled everywhere,** including a CSV first line `# DRY RUN — made-up data`.
- Endpoint facts are as listed in the spec's table (verified 2026-10-25). The People Also Ask question field is **unconfirmed**: read `title` of each nested element, and treat anything else as no question.
- Errors and logs never include credentials.

```python
"""Provider response cache: a repeat request within its TTL costs nothing.

Callers must not route dry-run providers through here -- made-up data must
never be served later as if it were real.
"""

from __future__ import annotations

import hashlib
import json
from datetime import datetime, timedelta
from typing import Any, Callable, Optional

from sqlalchemy import select
from sqlalchemy.exc import IntegrityError

from app.db.engine import get_session_factory
from app.db.models import ProviderCache, as_utc, utcnow


def request_hash(request: dict) -> str:
    canonical = json.dumps(request, sort_keys=True, separators=(",", ":"), ensure_ascii=False)
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


def cached_call(provider: str, endpoint: str, request: dict, ttl: timedelta,
                fetch: Callable[[], Any], factory=None, now: Optional[datetime] = None) -> Any:
    factory = factory or get_session_factory()
    now = now or utcnow()
    key = request_hash(request)
    where = (ProviderCache.provider == provider, ProviderCache.endpoint == endpoint,
             ProviderCache.request_hash == key)

    with factory() as session:
        row = session.scalar(select(ProviderCache).where(*where))
        if row is not None and as_utc(row.fetched_at) >= now - ttl:
            return row.response_json

    value = fetch()   # raises on failure: nothing is stored

    with factory.begin() as session:
        row = session.scalar(select(ProviderCache).where(*where))
        if row is None:
            try:
                with session.begin_nested():
                    session.add(ProviderCache(provider=provider, endpoint=endpoint, request_hash=key,
                                              response_json=value, fetched_at=now))
            except IntegrityError:
                pass  # a concurrent caller stored it first
        else:
            row.response_json, row.fetched_at = value, now
    return value
```

```python
SERP_PATH = "/v3/serp/google/organic/live/advanced"
ENDPOINT = BASE_URL + SERP_PATH
MAX_DEPTH = 200
OK = 20000

# ISO 3166-1 alpha-2 -> DataForSEO location_name. Extend as projects need.
LOCATIONS: Dict[str, str] = {
    "US": "United States", "GB": "United Kingdom", "IN": "India", "CA": "Canada",
    "AU": "Australia", "IE": "Ireland", "NZ": "New Zealand", "ZA": "South Africa",
    "SG": "Singapore", "AE": "United Arab Emirates", "DE": "Germany", "FR": "France",
    "ES": "Spain", "IT": "Italy", "NL": "Netherlands", "BR": "Brazil", "MX": "Mexico",
    "JP": "Japan",
}


def location_for(country: str) -> str:
    location = LOCATIONS.get(country.upper())
    if location is None:
        raise ValueError(f"no DataForSEO location mapped for country {country!r}")
    return location


def post_task(http, auth, path: str, task: dict) -> dict:
    """POST one task; return it, or raise ProviderError with DataForSEO's own
    status. Shared by every DataForSEO endpoint we call."""
    response = http.post(BASE_URL + path, json=[task], auth=auth, timeout=60)
    try:
        data = response.json()
    except ValueError:
        raise ProviderError(f"DataForSEO returned HTTP {response.status_code} without JSON")
    if data.get("status_code") != OK:
        raise ProviderError(f"DataForSEO {data.get('status_code')}: {data.get('status_message')}")
    tasks = data.get("tasks") or []
    if not tasks or tasks[0].get("status_code") != OK:
        first = tasks[0] if tasks else {}
        raise ProviderError(f"DataForSEO task {first.get('status_code')}: {first.get('status_message')}")
    return tasks[0]


def _questions(items) -> List[str]:
    """People Also Ask. The docs don't name the question field; each nested
    element's `title` is read, and anything else is ignored."""
    found = []
    for block in items:
        if block.get("type") != "people_also_ask":
            continue
        for element in block.get("items") or []:
            title = element.get("title") if isinstance(element, dict) else None
            if isinstance(title, str) and title.strip():
                found.append(title.strip())
    return found


class DataForSeoProvider:
    name = "dataforseo"

    def __init__(self, login: str, password: str, http=requests) -> None:
        self._auth = (login, password)
        self._http = http

    def search(self, query: SerpQuery) -> SerpResult:
        task = {
            "keyword": query.keyword,
            "location_name": location_for(query.country),
            "language_code": query.language,
            "device": query.device,
            "depth": max(1, min(query.depth, MAX_DEPTH)),
        }
        results = post_task(self._http, self._auth, SERP_PATH, task).get("result") or []
        raw = (results[0].get("items") or []) if results else []
        items = sorted(
            (SerpItem(position=int(i["rank_group"]), url=i["url"],
                      domain=domain_of(i["url"]), title=i.get("title") or "")
             for i in raw if i.get("type") == "organic" and i.get("url") and i.get("rank_group")),
            key=lambda item: item.position,
        )
        return SerpResult(items=items, provider=self.name, questions=_questions(raw))
```

```python
@dataclass(frozen=True)
class KeywordMetrics:
    keyword: str
    search_volume: Optional[int]
    cpc: Optional[float]
    competition: Optional[str]


@dataclass(frozen=True)
class RankedKeyword:
    keyword: str
    position: int
    url: str
    search_volume: Optional[int]
    difficulty: Optional[int]


class KeywordDataProvider(Protocol):
    name: str

    def metrics(self, keywords: List[str], country: str, language: str) -> Dict[str, KeywordMetrics]: ...

    def difficulty(self, keywords: List[str], country: str, language: str) -> Dict[str, int]: ...

    def ranked_keywords(self, domain: str, country: str, language: str, limit: int) -> List[RankedKeyword]: ...


def normalize_keyword(phrase: str) -> str:
    return " ".join(phrase.lower().split())


def clean_keywords(keywords: List[str]) -> Tuple[List[str], List[str]]:
    kept, dropped, seen = [], [], set()
    for raw in keywords:
        keyword = normalize_keyword(raw)
        if not keyword or keyword in seen:
            continue
        seen.add(keyword)
        (dropped if len(keyword) > MAX_KEYWORD_CHARS else kept).append(keyword)
    return kept, dropped
```

```python
"""DataForSEO keyword data: Google Ads search volume, Labs keyword difficulty,
Labs ranked keywords (shapes per docs.dataforseo.com, checked 2026-10-25).

Each call: cache first, then the dataforseo_labs budget, then HTTP -- so a
repeat within the TTL makes no request and spends nothing.
"""

from __future__ import annotations

from datetime import timedelta
from typing import Dict, List

import requests

from app.providers import cache, spend
from app.providers.dataforseo import location_for, post_task
from app.providers.keywords import KeywordMetrics, RankedKeyword, clean_keywords

BUDGET = "dataforseo_labs"
VOLUME_PATH = "/v3/keywords_data/google_ads/search_volume/live"
DIFFICULTY_PATH = "/v3/dataforseo_labs/google/bulk_keyword_difficulty/live"
RANKED_PATH = "/v3/dataforseo_labs/google/ranked_keywords/live"
METRICS_TTL = timedelta(days=30)
RANKED_TTL = timedelta(days=7)
MAX_RANKED_LIMIT = 1000


class DataForSeoKeywordProvider:
    name = "dataforseo"

    def __init__(self, login: str, password: str, cap: int, http=requests, factory=None) -> None:
        self._auth = (login, password)
        self._http = http
        self.cap = cap
        self._factory = factory

    def _call(self, path: str, task: dict, ttl: timedelta):
        def fetch():
            spend.reserve(BUDGET, self.cap, factory=self._factory)
            return post_task(self._http, self._auth, path, task).get("result") or []
        return cache.cached_call(BUDGET, path, task, ttl, fetch, factory=self._factory)

    def metrics(self, keywords: List[str], country: str, language: str) -> Dict[str, KeywordMetrics]:
        kept, _ = clean_keywords(keywords)
        if not kept:
            return {}
        task = {"keywords": kept, "location_name": location_for(country), "language_code": language}
        out = {}
        for item in self._call(VOLUME_PATH, task, METRICS_TTL):
            keyword = (item.get("keyword") or "").lower()
            if keyword:
                out[keyword] = KeywordMetrics(keyword, item.get("search_volume"),
                                              item.get("cpc"), item.get("competition"))
        return out

    def difficulty(self, keywords: List[str], country: str, language: str) -> Dict[str, int]:
        kept, _ = clean_keywords(keywords)
        if not kept:
            return {}
        task = {"keywords": kept, "location_name": location_for(country), "language_code": language}
        result = self._call(DIFFICULTY_PATH, task, METRICS_TTL)
        items = (result[0].get("items") or []) if result else []
        return {(i.get("keyword") or "").lower(): i["keyword_difficulty"]
                for i in items if i.get("keyword") and i.get("keyword_difficulty") is not None}

    def ranked_keywords(self, domain: str, country: str, language: str, limit: int) -> List[RankedKeyword]:
        task = {"target": domain, "location_name": location_for(country), "language_code": language,
                "limit": max(1, min(limit, MAX_RANKED_LIMIT)), "item_types": ["organic"]}
        result = self._call(RANKED_PATH, task, RANKED_TTL)
        items = (result[0].get("items") or []) if result else []
        out = []
        for item in items:
            data = item.get("keyword_data") or {}
            serp_item = (item.get("ranked_serp_element") or {}).get("serp_item") or {}
            keyword, position = data.get("keyword"), serp_item.get("rank_group")
            if not keyword or not position:
                continue
            out.append(RankedKeyword(
                keyword=keyword.lower(), position=int(position), url=serp_item.get("url") or "",
                search_volume=(data.get("keyword_info") or {}).get("search_volume"),
                difficulty=(data.get("keyword_properties") or {}).get("keyword_difficulty"),
            ))
        return out
```

```python
"""Explore a seed keyword: ideas, volume, difficulty, a traffic estimate, and
optional clusters of keywords whose search results overlap."""

from __future__ import annotations

import logging
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Tuple

from app.providers import registry
from app.providers.dry_run import DryRunProvider
from app.providers.keywords import normalize_keyword
from app.providers.serp import ProviderSkipped, SerpQuery, SerpResult

logger = logging.getLogger("seo_auditor")

# Share of a keyword's searches that the #1 organic result receives, as a round
# figure from published CTR studies. The traffic figure is an *estimate* from
# this constant, not measured traffic -- and is labelled so.
CTR_TOP1 = 0.28
MAX_IDEAS = 50
MAX_CLUSTER = 20
MIN_SHARED_URLS = 3


@dataclass
class ExploreResult:
    seed: str
    country: str
    language: str
    dry_run: bool
    keywords: List[Dict] = field(default_factory=list)
    clusters: List[List[str]] = field(default_factory=list)
    notes: List[str] = field(default_factory=list)


def _resolve_serp(data):
    """Dry-run data previews with dry-run SERPs; live data uses a live SERP
    provider if one exists, else none (no PAA, no clustering)."""
    if not registry.is_live(data):
        return DryRunProvider()
    try:
        serp = registry.get_serp_provider()
    except ProviderSkipped:
        return None
    return serp if registry.is_live(serp) else None


def cluster_keywords(keywords: List[str], serp, country: str, language: str,
                     prefetched: Optional[Dict[str, SerpResult]] = None) -> Tuple[List[List[str]], List[str]]:
    prefetched = prefetched or {}
    tops: Dict[str, set] = {}
    notes: List[str] = []
    for keyword in keywords[:MAX_CLUSTER]:
        try:
            result = prefetched.get(keyword) or serp.search(SerpQuery(keyword, country, language, depth=10))
        except ProviderSkipped as exc:
            notes.append(f"{keyword!r} not clustered: {exc}")
            continue
        except Exception as exc:
            notes.append(f"{keyword!r} not clustered: {type(exc).__name__}")
            continue
        tops[keyword] = {item.url for item in result.items[:10]}

    parent = {k: k for k in tops}

    def find(k):
        while parent[k] != k:
            parent[k] = parent[parent[k]]
            k = parent[k]
        return k

    checked = list(tops)
    for i, a in enumerate(checked):
        for b in checked[i + 1:]:
            if len(tops[a] & tops[b]) >= MIN_SHARED_URLS:
                parent[find(a)] = find(b)

    groups: Dict[str, List[str]] = {}
    for k in checked:
        groups.setdefault(find(k), []).append(k)
    order = {k: i for i, k in enumerate(keywords)}   # callers pass keywords volume-sorted
    clusters = [sorted(g, key=order.get) for g in groups.values()]
    return sorted(clusters, key=lambda g: order[g[0]]), notes


def explore(seed: str, country: str = "US", language: str = "en", cluster: bool = False,
            serp=None, data=None) -> ExploreResult:
    from app.seo import keyword_suggestions   # looked up at call time: tests replace it

    seed = normalize_keyword(seed)
    if not seed:
        raise ValueError("seed keyword is empty")
    data = data or registry.get_keyword_provider()
    if serp is None:
        serp = _resolve_serp(data)
    result = ExploreResult(seed=seed, country=country.upper(), language=language.lower(),
                           dry_run=not registry.is_live(data))

    sources: Dict[str, str] = {}

    def add(phrase, source):
        keyword = normalize_keyword(phrase)
        if keyword and keyword != seed and keyword not in sources:
            sources[keyword] = source

    for phrase in keyword_suggestions._autocomplete_suggestions(seed):
        add(phrase, "autocomplete")

    prefetched: Dict[str, SerpResult] = {}
    if serp is not None:
        try:
            prefetched[seed] = serp.search(SerpQuery(seed, result.country, result.language, depth=10))
            for question in prefetched[seed].questions:
                add(question, "people_also_ask")
        except ProviderSkipped as exc:
            result.notes.append(f"People Also Ask skipped: {exc}")
        except Exception as exc:
            result.notes.append(f"People Also Ask failed: {type(exc).__name__}")

    ideas = list(sources)[:MAX_IDEAS]
    keywords = [seed] + ideas

    metrics, difficulty = {}, {}
    try:
        metrics = data.metrics(keywords, result.country, result.language)
        difficulty = data.difficulty(keywords, result.country, result.language)
    except ProviderSkipped:
        raise
    except Exception as exc:
        logger.error("Keyword metrics failed for %r: %s", seed, type(exc).__name__)
        result.notes.append(f"keyword metrics unavailable: {type(exc).__name__}")

    rows = []
    for keyword in keywords:
        m = metrics.get(keyword)
        volume = m.search_volume if m else None
        rows.append({
            "keyword": keyword,
            "source": "seed" if keyword == seed else sources[keyword],
            "search_volume": volume,
            "cpc": m.cpc if m else None,
            "competition": m.competition if m else None,
            "difficulty": difficulty.get(keyword),
            "traffic_potential_estimate": round(volume * CTR_TOP1) if volume is not None else None,
        })
    rows.sort(key=lambda r: (r["search_volume"] is None, -(r["search_volume"] or 0), r["keyword"]))
    result.keywords = rows

    if cluster:
        if serp is None:
            result.notes.append("clustering needs a live SERP provider")
        else:
            result.clusters, notes = cluster_keywords([r["keyword"] for r in rows], serp,
                                                      result.country, result.language, prefetched)
            result.notes.extend(notes)
    return result
```

```python
from dataclasses import dataclass, field
from typing import Dict, List, Optional

from app.db.projects import project_domain
from app.providers import registry

MAX_COMPETITORS = 5
COMPETITOR_MAX_POSITION = 20
MINE_MAX_POSITION = 100
UNKNOWN_DIFFICULTY = 50


def opportunity(volume: Optional[int], difficulty: Optional[int]) -> float:
    kd = UNKNOWN_DIFFICULTY if difficulty is None else difficulty
    return round((volume or 0) * (100 - kd) / 100, 1)


@dataclass
class GapResult:
    my_domain: str
    competitors: List[str]
    dry_run: bool
    rows: List[Dict] = field(default_factory=list)
    total_gaps: int = 0


def content_gap(my_domain: str, competitors: List[str], country: str = "US", language: str = "en",
                limit: int = 500, data=None) -> GapResult:
    mine_domain = project_domain(my_domain)
    rivals: List[str] = []
    for c in competitors:
        d = project_domain(c)
        if d and d != mine_domain and d not in rivals:
            rivals.append(d)
    if not 1 <= len(rivals) <= MAX_COMPETITORS:
        raise ValueError(f"need 1 to {MAX_COMPETITORS} competitors other than {mine_domain}")

    data = data or registry.get_keyword_provider()
    country, language = country.upper(), language.lower()
    mine = {rk.keyword: rk.position for rk in data.ranked_keywords(mine_domain, country, language, limit)}

    rows: Dict[str, Dict] = {}
    for rival in rivals:
        for rk in data.ranked_keywords(rival, country, language, limit):
            if rk.position > COMPETITOR_MAX_POSITION:
                continue
            if mine.get(rk.keyword, MINE_MAX_POSITION + 1) <= MINE_MAX_POSITION:
                continue
            row = rows.setdefault(rk.keyword, {"keyword": rk.keyword, "search_volume": rk.search_volume,
                                               "difficulty": rk.difficulty, "competitors": {}})
            if row["search_volume"] is None:
                row["search_volume"] = rk.search_volume
            if row["difficulty"] is None:
                row["difficulty"] = rk.difficulty
            row["competitors"][rival] = {"position": rk.position, "url": rk.url}

    finished = []
    for row in rows.values():
        row["best_competitor_position"] = min(c["position"] for c in row["competitors"].values())
        row["opportunity"] = opportunity(row["search_volume"], row["difficulty"])
        finished.append(row)
    finished.sort(key=lambda r: (-r["opportunity"], -(r["search_volume"] or 0), r["keyword"]))
    return GapResult(my_domain=mine_domain, competitors=rivals, dry_run=not registry.is_live(data),
                     rows=finished, total_gaps=len(finished))
```

```python
"""Keywords Explorer and Content Gap over HTTP, as JSON and CSV."""

from __future__ import annotations

import csv
import io
from dataclasses import asdict
from typing import List, Optional

from fastapi import APIRouter, HTTPException, Query, Response
from pydantic import BaseModel, Field
from sqlalchemy import select

from app.api.projects import _factory
from app.db.models import Competitor, Project
from app.keywords import explorer, gap
from app.providers.serp import ProviderError, ProviderSkipped

router = APIRouter()
DRY_RUN_LINE = "# DRY RUN — made-up data"


class ExploreIn(BaseModel):
    seed: str = Field(max_length=80)
    country: str = Field("US", min_length=2, max_length=2)
    language: str = Field("en", min_length=2, max_length=8)
    cluster: bool = False


class GapIn(BaseModel):
    competitors: Optional[List[str]] = None
    country: str = Field("US", min_length=2, max_length=2)
    language: str = Field("en", min_length=2, max_length=8)
    limit: int = Field(500, ge=10, le=1000)


def _explore(seed, country, language, cluster):
    try:
        return explorer.explore(seed, country=country, language=language, cluster=cluster)
    except ProviderSkipped as exc:
        raise HTTPException(status_code=429, detail=str(exc))
    except ProviderError as exc:
        raise HTTPException(status_code=502, detail=str(exc))
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc))


def _csv(dry_run: bool, header: List[str], rows: List[List]) -> str:
    out = io.StringIO()
    if dry_run:
        out.write(DRY_RUN_LINE + "\n")
    writer = csv.writer(out, lineterminator="\n")
    writer.writerow(header)
    writer.writerows(rows)
    return out.getvalue()


def _csv_response(text: str, filename: str) -> Response:
    return Response(content=text, media_type="text/csv; charset=utf-8",
                    headers={"Content-Disposition": f'attachment; filename="{filename}"'})


@router.post("/keywords/explore")
def explore_keywords(body: ExploreIn):
    return asdict(_explore(body.seed, body.country, body.language, body.cluster))


@router.get("/keywords/explore.csv")
def explore_keywords_csv(seed: str = Query(..., max_length=80), country: str = "US", language: str = "en"):
    result = _explore(seed, country, language, False)
    columns = ["keyword", "source", "search_volume", "cpc", "competition", "difficulty",
               "traffic_potential_estimate"]
    rows = [[k[c] if k[c] is not None else "" for c in columns] for k in result.keywords]
    return _csv_response(_csv(result.dry_run, columns, rows), "keywords.csv")


def _gap(project_id: int, competitors: Optional[List[str]], country: str, language: str, limit: int):
    with _factory()() as session:
        project = session.get(Project, project_id)
        if project is None:
            raise HTTPException(status_code=404, detail="no such project")
        domain = project.domain
        if not competitors:
            competitors = list(session.scalars(select(Competitor.domain)
                                               .where(Competitor.project_id == project_id)
                                               .order_by(Competitor.domain)))
    if not competitors:
        raise HTTPException(status_code=400, detail="add competitors to the project, or pass some")
    try:
        return gap.content_gap(domain, competitors, country=country, language=language, limit=limit)
    except ProviderSkipped as exc:
        raise HTTPException(status_code=429, detail=str(exc))
    except ProviderError as exc:
        raise HTTPException(status_code=502, detail=str(exc))
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc))


@router.post("/projects/{project_id}/content-gap")
def content_gap(project_id: int, body: GapIn):
    return asdict(_gap(project_id, body.competitors, body.country, body.language, body.limit))


@router.get("/projects/{project_id}/content-gap.csv")
def content_gap_csv(project_id: int, competitor: Optional[List[str]] = Query(None),
                    country: str = "US", language: str = "en", limit: int = Query(500, ge=10, le=1000)):
    result = _gap(project_id, competitor, country, language, limit)
    header = ["keyword", "search_volume", "difficulty", "opportunity", "best_competitor_position"]
    for rival in result.competitors:
        header += [f"{rival} position", f"{rival} url"]
    rows = []
    for row in result.rows:
        line = [row["keyword"], row["search_volume"] or "", "" if row["difficulty"] is None else row["difficulty"],
                row["opportunity"], row["best_competitor_position"]]
        for rival in result.competitors:
            hit = row["competitors"].get(rival)
            line += [hit["position"], hit["url"]] if hit else ["", ""]
        rows.append(line)
    return _csv_response(_csv(result.dry_run, header, rows), f"content-gap-{result.my_domain}.csv")
```
